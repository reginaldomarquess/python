# 4 tipos primitivos mais básicos

int  = números inteiro -> 7, -4, 0 9875
floa = numeros reais ou de pontos flutuantes -> 4.1... 0.007... 7.0
bool = valores lógicos ou booleanos -> Só aceita True ou False
str  = valores de carcteres ou de strings -> 'Olá', '7.5', ou string fazia ''

# pow = potência
# sqrt = calcular a raiz quadrada
# factorial = calcular o fatorial de um número

Manipulação de Strings

 Fatiamento de String;
# len(); comprimento, usado para saber o comprimento de uma string
# count(); serve para contar cacteres específicos dentro de uma string
# find(); encontrar, serve para encontrar pedaços específicos que forem informados na função
# in; serve para saber se existe uma palavra específica dentro de uma cadeia de caracteres
 Transformação:
# replace(); trocar, reposicionar, reposiciona uma string ou cadeia de caracteres
# upper(); transforma todos os caracteres em maiúsculas
# lower(); transforma todos os caracteres me minúsculas
# capitalize(); joga todos os caracteres para minúsculos, porém os primeiros cacteres ficam maiúsculos
# title(); analisa quantas palavras a cadeira de caractere tem e "joga" todos os primeiros caracteres para maíusculas 
# strip(); remove todos os espaços tanto no início quanto no fianl
# rtrip(); remove somente os espaços à esquerda, ou seja, os espaços do final da string/frase
# lstrip(); remove os espeços somente da esquerda e mantem os espaços da direita
 Divisão:
# split(); divide toda a frase em uma nova lista onde cada elemento é separado pelo seu espaço
 Junção:
# join(); junta todos os elementos de uma cadeia de caracteres e coloca um "traço" onde haviam espaços
